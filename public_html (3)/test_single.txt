<script>\r\n
    (function () {\r\n
        const form = document.getElementById('bookingForm');\r\n
        const steps = Array.from(document.querySelectorAll('[data-step-pane]'));\r\n
        const indicators = Array.from(document.querySelectorAll('[data-step-indicator]'));\r\n
        let currentStep = 1;\r\n
\r\n
        function showStep(step) {\r\n
            steps.forEach(function (pane) {\r\n
                pane.classList.toggle('is-active', Number(pane.dataset.stepPane) === step);\r\n
            });\r\n
            indicators.forEach(function (indicator) {\r\n
                const num = Number(indicator.dataset.stepIndicator);\r\n
                const circle = indicator.querySelector('.booking-step-circle');\r\n
                indicator.classList.toggle('is-active', num === step);\r\n
                indicator.classList.toggle('is-complete', num < step);\r\n
                if (circle) {\r\n
                    if (num === step) {\r\n
                        circle.setAttribute('aria-current', 'step');\r\n
                    } else {\r\n
                        circle.removeAttribute('aria-current');\r\n
                    }\r\n
                }\r\n
            });\r\n
            currentStep = step;\r\n
            if (step === steps.length) {\r\n
                window.requestAnimationFrame(updateSummary);\r\n
                window.setTimeout(updateSummary, 0);\r\n
            }\r\n
            window.scrollTo({ top: form.offsetTop - 120, behavior: 'smooth' });\r\n
        }\r\n
\r\n
        function validateStep(step) {\r\n
            const pane = steps.find(function (p) { return Number(p.dataset.stepPane) === step; });\r\n
            const fields = pane.querySelectorAll('input[required], select[required]');\r\n
            const actionArea = pane.querySelector('.booking-pane-actions') || pane.querySelector('.booking-pane-actions-split');\r\n
            const nextButton = pane.querySelector('.booking-next-btn, .booking-submit-btn');\r\n
            const inlineMessage = pane.querySelector('.booking-inline-error') || document.createElement('div');\r\n
            let valid = true;\r\n
\r\n
            inlineMessage.className = 'booking-inline-error';\r\n
            inlineMessage.setAttribute('role', 'alert');\r\n
            inlineMessage.setAttribute('aria-live', 'polite');\r\n
            inlineMessage.textContent = 'Please fill in all required fields before continuing.';\r\n
            inlineMessage.style.display = 'none';\r\n
            inlineMessage.style.marginBottom = '12px';\r\n
            inlineMessage.style.padding = '8px 12px';\r\n
            inlineMessage.style.borderRadius = '4px';\r\n
            inlineMessage.style.background = '#fdecec';\r\n
            inlineMessage.style.color = '#a11c1c';\r\n
            inlineMessage.style.border = '1px solid #f5b7b7';\r\n
            inlineMessage.style.fontSize = '14px';\r\n
\r\n
            if (!inlineMessage.parentNode && actionArea) {\r\n
                actionArea.insertBefore(inlineMessage, nextButton || actionArea.firstChild);\r\n
            }\r\n
\r\n
            function clearInvalidState(event) {\r\n
                const field = event.target;\r\n
                if (field) {\r\n
                    field.classList.remove('is-invalid');\r\n
                    field.setAttribute('aria-invalid', 'false');\r\n
                    if (field.value.trim()) {\r\n
                        field.classList.remove('is-invalid');\r\n
                        field.setAttribute('aria-invalid', 'false');\r\n
                    }\r\n
                }\r\n
                if (pane.querySelectorAll('.is-invalid').length === 0) {\r\n
                    inlineMessage.style.display = 'none';\r\n
                }\r\n
            }\r\n
\r\n
            fields.forEach(function (field) {\r\n
                field.addEventListener('input', clearInvalidState);\r\n
                field.addEventListener('change', clearInvalidState);\r\n
\r\n
                if (field.type === 'radio') {\r\n
                    const group = pane.querySelectorAll('input[name="' + field.name + '"]');\r\n
                    const checked = Array.from(group).some(function (r) { return r.checked; });\r\n
                    if (!checked) {\r\n
                        field.setAttribute('aria-invalid', 'true');\r\n
                        valid = false;\r\n
                    }\r\n
                } else if (!field.value.trim()) {\r\n
                    field.classList.add('is-invalid');\r\n
                    field.setAttribute('aria-invalid', 'true');\r\n
                    valid = false;\r\n
                } else {\r\n
                    field.classList.remove('is-invalid');\r\n
                    field.setAttribute('aria-invalid', 'false');\r\n
                }\r\n
            });\r\n
\r\n
            if (!valid) {\r\n
                inlineMessage.style.display = 'block';\r\n
            } else {\r\n
                inlineMessage.style.display = 'none';\r\n
            }\r\n
            return valid;\r\n
        }\r\n
\r\n
        document.querySelectorAll('.booking-next-btn').forEach(function (btn) {\r\n
            btn.addEventListener('click', function () {\r\n
                if (!validateStep(currentStep)) return;\r\n
                if (currentStep < steps.length) showStep(currentStep + 1);\r\n
            });\r\n
        });\r\n
\r\n
        document.querySelectorAll('.booking-back-btn').forEach(function (btn) {\r\n
            btn.addEventListener('click', function () {\r\n
                if (currentStep > 1) showStep(currentStep - 1);\r\n
            });\r\n
        });\r\n
\r\n
        form.addEventListener('submit', function () {\r\n
            const submitButton = form.querySelector('.booking-submit-btn');\r\n
            if (!submitButton) return;\r\n
            submitButton.disabled = true;\r\n
            submitButton.innerHTML = '<span class="spinner-border spinner-border-sm me-2" role="status" aria-hidden="true"></span><span>Booking...</span>';\r\n
        });\r\n
\r\n
        function updateSummary() {\r\n
            const selectedService = form.querySelector('input[name="service_id"]:checked');\r\n
            const valueOf = function (name) {\r\n
                const field = form.elements.namedItem(name);\r\n
                const value = field && typeof field.value === 'string' ? field.value.trim() : '';\r\n
                return value || '-';\r\n
            };\r\n
\r\n
            document.getElementById('summaryService').textContent = selectedService ? selectedService.dataset.name : '-';\r\n
            document.getElementById('summaryVehicle').textContent = valueOf('vehicle_make_model');\r\n
            document.getElementById('summaryReg').textContent = valueOf('registration_number');\r\n
\r\n
            const dateField = form.elements.namedItem('appointment_date');\r\n
            const timeField = form.elements.namedItem('appointment_time');\r\n
            const date = dateField && typeof dateField.value === 'string' ? dateField.value.trim() : '';\r\n
            const time = timeField && typeof timeField.value === 'string' ? timeField.value.trim() : '';\r\n
            document.getElementById('summaryDateTime').textContent = date && time ? (date + ' ┬╖ ' + time) : '-';\r\n
\r\n
            document.getElementById('summaryName').textContent = valueOf('customer_name');\r\n
            document.getElementById('summaryPhone').textContent = valueOf('customer_phone');\r\n
            document.getElementById('summaryEmail').textContent = valueOf('customer_email');\r\n
            document.getElementById('summaryContactMethod').textContent = valueOf('preferred_contact_method');\r\n
            document.getElementById('summaryReason').textContent = valueOf('service_reason');\r\n
            document.getElementById('summaryIssues').textContent = valueOf('additional_issues');\r\n
        }\r\n
\r\n
        form.addEventListener('input', function () {\r\n
            if (currentStep === steps.length) updateSummary();\r\n
        });\r\n
\r\n
        form.addEventListener('change', function () {\r\n
            if (currentStep === steps.length) updateSummary();\r\n
        });\r\n
\r\n
        function updateSelectedService() {\r\n
            const selectedService = form.querySelector('input[name="service_id"]:checked');\r\n
            const title = document.getElementById('selectedServiceTitle');\r\n
            const description = document.getElementById('selectedServiceDescription');\r\n
            const triggerName = document.getElementById('serviceTriggerName');\r\n
            const editorName = document.getElementById('editorServiceName');\r\n
\r\n
            if (!selectedService) return;\r\n
\r\n
            const serviceName = selectedService.dataset.name || 'Selected service';\r\n
            title.textContent = serviceName;\r\n
            triggerName.textContent = serviceName;\r\n
            description.textContent = 'Professional ' + serviceName.toLowerCase() + ' support from the United Auto workshop team.';\r\n
            editorName.textContent = serviceName;\r\n
        }\r\n
\r\n
        form.querySelectorAll('input[name="service_id"]').forEach(function (serviceInput) {\r\n
            serviceInput.addEventListener('change', updateSelectedService);\r\n
        });\r\n
\r\n
        const servicePicker = document.querySelector('.booking-service-picker');\r\n
        const serviceTrigger = document.querySelector('.booking-service-trigger');\r\n
\r\n
        serviceTrigger.addEventListener('click', function () {\r\n
            const isOpen = servicePicker.classList.toggle('is-open');\r\n
            if (isOpen) {\r\n
                const triggerBottom = serviceTrigger.getBoundingClientRect().bottom;\r\n
                servicePicker.classList.toggle('opens-up', triggerBottom + 290 > window.innerHeight);\r\n
            } else {\r\n
                servicePicker.classList.remove('opens-up');\r\n
            }\r\n
            serviceTrigger.setAttribute('aria-expanded', String(isOpen));\r\n
        });\r\n
\r\n
        servicePicker.querySelectorAll('.service-option').forEach(function (option) {\r\n
            option.addEventListener('click', function () {\r\n
                window.setTimeout(function () {\r\n
                    servicePicker.classList.remove('is-open');\r\n
                    servicePicker.classList.remove('opens-up');\r\n
                    serviceTrigger.setAttribute('aria-expanded', 'false');\r\n
                    updateSelectedService();\r\n
                }, 0);\r\n
            });\r\n
        });\r\n
\r\n
        document.addEventListener('click', function (event) {\r\n
            if (!servicePicker.contains(event.target)) {\r\n
                servicePicker.classList.remove('is-open');\r\n
                servicePicker.classList.remove('opens-up');\r\n
                serviceTrigger.setAttribute('aria-expanded', 'false');\r\n
            }\r\n
        });\r\n
